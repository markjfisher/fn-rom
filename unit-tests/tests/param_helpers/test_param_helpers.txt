marker $$ num_params returns zero for empty tail $$

registers flush
run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)
assert CP = $end_init_harness $$completed harness init$$
run $t_num_params_zero until CP = $t_num_params_zero_end
assert A = 0x00 $$no params counted$$
