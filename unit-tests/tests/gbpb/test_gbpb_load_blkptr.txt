///////////////////////////////
marker $$ block at &0A00 (X=&00, Y=&0A) $$
///////////////////////////////

memory load #0x8000 "${WS_ROOT}/build/fujinet.rom"
symbols load "${WS_ROOT}/build/fujinet.rom.lbl"

memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"

symbols remove "__.*"

registers flush

memory write $fuji_gbpbv_blk_save_ptr 0x(00,0A)
memory write $aws_tmp04 0x(EE,EE)

run $t_load until (CP = $t_load_end OR cycle_count > 1000)

assert CP = $t_load_end $$completed$$
assert $aws_tmp04 = 0x00 $$pointer low byte$$
assert $aws_tmp05 = 0x0A $$pointer high byte$$
