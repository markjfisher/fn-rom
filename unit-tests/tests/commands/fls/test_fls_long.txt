///////////////////////////////
marker $$ test LONG keyword parse $$
///////////////////////////////

memory load rom #0x8000 "${WS_ROOT}/build/fujinet.rom"
symbols load "${WS_ROOT}/build/fujinet.rom.lbl"
memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
symbols remove "__.*"
memory protect ro #0xFF00 0x0100
registers flush
run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)

memory write $fuji_filename_len 0x(04)
memory write #0x1000 0x(4c,4f,4e,47,00)
registers set cycle_count=0
run $cfl_parse_long_kw until (cycle_count > 500)
assert $fuji_channel_scratch = 0x00 $$not set until command stores it$$
assert $pws_tmp02 = 0xFF $$parsed LONG$$

memory write $fuji_filename_len 0x(03)
memory write #0x1000 0x(4c,34,30,00)
memory write $pws_tmp02 0x(00)
registers set cycle_count=0
run $cfl_parse_long_kw until (cycle_count > 500)
assert $pws_tmp02 = 0x00 $$L40 is not LONG$$

memory write $fuji_filename_len 0x(04)
memory write #0x1000 0x(4c,4f,4e,00)
memory write $pws_tmp02 0x(00)
registers set cycle_count=0
run $cfl_parse_long_kw until (cycle_count > 500)
assert $pws_tmp02 = 0x00 $$LON is not LONG$$
