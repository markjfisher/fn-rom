marker $$ test *FLS LONG listing output $$

; --------------------------------------------------
; load fujinet rom
memory load rom #0x8000 "${WS_ROOT}/build/fujinet.rom"
symbols load "${WS_ROOT}/build/fujinet.rom.lbl"

memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"

symbols remove "__.*"

memory protect ro #0xFF00 0x0100

registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)
assert CP = $end_init_harness $$completed code correctly$$
assert cycle_count <= 2000    $$cycles within accpeptable limit from previous runs$$
; --------------------------------------------------

memory fill #0xC800~#0xCFFF   $$ clear print capture region $$

; Decoded ListDirectory payload (from 003 SLIP capture).
memory load #0x1700 "${UNIT_TEST_DIR}/data/fls/003_list_directory_root_long_decoded.bin"

; First entry in decoded blob (flags, len, name at $170A).
memory write $aws_tmp00 0x(0a,17)
memory write $cws_tmp8 0x(01)
memory write $cws_tmp1 0x(05)
memory write $cws_tmp2 0x(01)
memory write $cws_tmp3 0x(00)
memory write $fuji_channel_scratch 0x(01)

memory write $mock_print_pos 0x(00)
memory write $mock_print_armed 0x(01)

registers set cycle_count=0
run $cfl_long_entry until (cycle_count > 50000)

memory show #0xC800 0x40

assert #0xC800 ~ "DIR 4096 \x3f " $$long listing first line (mtime > 32-bit prints ?)$$
assert #0xC80B ~ "atari" $$first entry name$$

///////////////////////////////
marker $$ test *FLS LONG via SLIP replay $$
///////////////////////////////

memory load rom #0x8000 "${WS_ROOT}/build/fujinet.rom"
symbols load "${WS_ROOT}/build/fujinet.rom.lbl"
memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
symbols remove "__.*"
memory protect ro #0xFF00 0x0100
registers flush
run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)
assert CP = $end_init_harness $$completed code correctly$$

memory fill #0xC800~#0xCFFF

memory load #0xC000 "${UNIT_TEST_DIR}/data/fls/003_list_directory_root_start0_maxpayload220_resp.bin"
memory write $mock_slip_len 0x(B9,00)
run $mock_slip_rewind until (cycle_count > 100)

memory write $fuji_current_host_len 0x(15)
memory write $fuji_current_fs_len 0x(15)
; Host URI at PWS+$168; FS URI scratch at PWS+$118 (see fujinet.inc).
memory write #0x1868 0x(74,6e,66,73,3a,2f,2f,31,39,32,2e,31,36,38,2e,31,2e,31,30,31,2f,00)
memory write #0x1818 0x(74,6e,66,73,3a,2f,2f,31,39,32,2e,31,36,38,2e,31,2e,31,30,31,2f,00)

memory write $mock_print_pos 0x(00)
memory write $mock_print_armed 0x(00)
memory write $fuji_channel_scratch 0x(01)
memory write $pws_tmp04 0x(00)
memory write $pws_tmp05 0x(00)

; Receive the ListDirectory response, then print entries (same data as *FLS LONG).
registers set cycle_count=0
run $fujibus_receive_packet while (#0x1B5C < 0xB9)

assert #0x1B5C = 0xB9 $$SLIP response fully replayed$$
assert $buffer_ptr -> #0x1700 $$PWS packet at page$$
assert #0x1700 = 0xFE $$decoded FujiBus device id$$
assert #0x1705 = 0x01 $$FujiBus status OK$$
assert #0x170D = 0x06 $$entry count low$$

memory write $fuji_channel_scratch 0x(01)
registers set cycle_count=0
run $cfl_rxlen_ok until (cycle_count > 5000)
assert $cws_tmp2 = 0x06 $$entry count from response header$$
assert $aws_tmp00 -> #0x1711 $$first entry ptr$$
assert $aws_tmp12 -> #0x17b7 $$entries blob end$$
assert #0x1711 = 0x01 $$first entry flags (dir)$$
assert #0x1713 ~ "atari" $$first entry name in PWS buffer$$

memory write $mock_print_pos 0x(00)
memory write $mock_print_armed 0x(01)
registers set cycle_count=0
run $cfl_entry_loop while ($mock_print_pos < 0x83)

assert $mock_print_pos = 0x83 $$six long lines (LF+CR each)$$
assert #0xC800 ~ "DIR 4096 \x3f atari\x0a\x0dDIR 4096 \x3f bbc\x0a\x0dDIR 4096 \x3f subdir\x0a\x0dFILE 18 \x3f test.txt\x0a\x0dFILE 55 \x3f tnfs-write-tcp.txt\x0a\x0dFILE 55 \x3f tnfs-write-udp.txt" $$long listing via SLIP$$
